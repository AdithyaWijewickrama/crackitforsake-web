        int u, v;

        g[u].push_back(v);
    }

    // state = node*2 + p ; p=0 -> next move must go UP, p=1 -> must go DOWN
    vector<int> d(2 * (n + 1), -1);
    queue<int> q;
    d[1 * 2 + 0] = 0;
    q.push(1 * 2 + 0);

        int s = q.front(); q.pop();
        int u = s >> 1, p = s & 1;
        for (int v : g[u]) {

            if (v == 1) {                 // closed the cycle

            if (d[t] < 0) {
                d[t] = d[s] + 1;

            }
        }
    }

#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
const ll MOD = 1000000007LL;
const ll INF = 1000000000000000000LL; // cost of x[0]=1 (forces x[0]=0)

int n, q, m;
vector<int> par_, dpar, sz_;
vector<ll> A, B;      // A: cost if root=0, B: cost if root=1
ll S; int E, bad;     // S = sum of min(A,B); E = #components with A==B

struct Rec { int rc, rr; ll oa, ob; int type; };
vector<Rec> st;

inline int findp(int x, int &p) {
    p = 0;
    while (par_[x] != x) { p ^= dpar[x]; x = par_[x]; }
    return x;
}
inline void rem(int r) { S -= min(A[r], B[r]); E -= (A[r] == B[r]); }
inline void addc(int r) { S += min(A[r], B[r]); E += (A[r] == B[r]); }

void unite(int u, int v, int b) {
    int pu, pv;
    int ru = findp(u, pu), rv = findp(v, pv);
    if (ru == rv) { if ((pu ^ pv) != b) bad++; return; }
    if (sz_[ru] < sz_[rv]) swap(ru, rv);
    int d = b ^ pu ^ pv;
    rem(ru); rem(rv);
    st.push_back({rv, ru, A[ru], B[ru], 0});
    par_[rv] = ru; dpar[rv] = d; sz_[ru] += sz_[rv];
    if (d == 0) { A[ru] += A[rv]; B[ru] += B[rv]; }
    else { ll na = A[ru] + B[rv], nb = B[ru] + A[rv]; A[ru] = na; B[ru] = nb; }
    addc(ru);
}
void addCost(int i, ll a, ll b) {
    int p; int r = findp(i, p);
    rem(r);
    st.push_back({r, 0, A[r], B[r], 1});
    if (p == 0) { A[r] += a; B[r] += b; } else { A[r] += b; B[r] += a; }
    addc(r);
}
void undoTo(size_t sp) {
    while (st.size() > sp) {
        Rec r = st.back(); st.pop_back();
        if (r.type == 0) {
            par_[r.rc] = r.rc; dpar[r.rc] = 0; sz_[r.rr] -= sz_[r.rc];
            A[r.rr] = r.oa; B[r.rr] = r.ob;
        } else { A[r.rc] = r.oa; B[r.rc] = r.ob; }
    }
}

// segment tree with linked lists of entries
vector<int> headE, headC, nxtE, valE, nxtC, valC;
vector<int> eu, ev, eb;
vector<ll> ca, cb; vector<int> ci;

void ins(vector<int> &head, vector<int> &nxt, vector<int> &val, int node, int l, int r, int ql, int qr, int idx) {
    if (ql > r || qr < l) return;
    if (ql <= l && r <= qr) { val.push_back(idx); nxt.push_back(head[node]); head[node] = (int)val.size() - 1; return; }
    int mid = (l + r) / 2;
    ins(head, nxt, val, node * 2, l, mid, ql, qr, idx);
    ins(head, nxt, val, node * 2 + 1, mid + 1, r, ql, qr, idx);
}

vector<int> ak; vector<array<int, 9>> at_;
vector<ll> ansS; vector<int> ansE; // ansS=-1 -> infeasible
vector<ll> pw2;

void dfs(int node, int l, int r) {
    ll sS = S; int sE = E, sb = bad; size_t sp = st.size();
    for (int e = headE[node]; e != -1; e = nxtE[e]) { int id = valE[e]; unite(eu[id], ev[id], eb[id]); }
    for (int e = headC[node]; e != -1; e = nxtC[e]) { int id = valC[e]; addCost(ci[id], ca[id], cb[id]); }
    if (bad > 0) {
        for (int i = l; i <= r; i++) ansS[i] = -1;
    } else if (l == r) {
        for (int j = 0; j < ak[l]; j++) unite(at_[l][3 * j], at_[l][3 * j + 1], at_[l][3 * j + 2]);
        if (bad > 0) ansS[l] = -1; else { ansS[l] = S; ansE[l] = E; }
    } else {
        int mid = (l + r) / 2;
        dfs(node * 2, l, mid);
        dfs(node * 2 + 1, mid + 1, r);
    }
    undoTo(sp); S = sS; E = sE; bad = sb;
}

import java.io.*;
import java.util.*;

public class Main {
    static final int B = 18;
    static int n;                 // leaves = N+1 (index 0 unused)
    static int[] t;               // flat segment tree: node p -> t[p*B .. p*B+B-1]
    static int[] E, fen;

    // ---------- fast input ----------
    static DataInputStream in = new DataInputStream(new BufferedInputStream(System.in, 1 << 16));
    static byte[] buf = new byte[1 << 16];
    static int bufLen = 0, bufPtr = 0;

    static int readByte() throws IOException {
        if (bufPtr == bufLen) {
            bufLen = in.read(buf, 0, buf.length);
            bufPtr = 0;
            if (bufLen <= 0) return -1;
        }
        return buf[bufPtr++];
    }

    static char readChar() throws IOException {
        int c = readByte();
        while (c <= ' ') c = readByte();
        return (char) c;
    }

    static long readLong() throws IOException {
        int c = readByte();
        while (c <= ' ') c = readByte();
        boolean neg = false;
        if (c == '-') { neg = true; c = readByte(); }
        long r = 0;
        while (c >= '0' && c <= '9') {
            r = r * 10 + (c - '0');
            c = readByte();
        }
        return neg ? -r : r;
    }

    // ---------- linear basis ----------
    static void ins(int[] b, int off, int x) {
        while (x != 0) {
            int h = 31 - Integer.numberOfLeadingZeros(x);
            if (b[off + h] == 0) { b[off + h] = x; return; }
            x ^= b[off + h];
        }
    }

    static void pull(int p) {
        int o = p * B, l = 2 * p * B, r = (2 * p + 1) * B;
        System.arraycopy(t, l, t, o, B);
        for (int i = 0; i < B; i++) {
            int v = t[r + i];
            if (v != 0) ins(t, o, v);
        }
    }

    // ---------- Fenwick (prefix XOR of E = A[i]) ----------
    static void fenUpd(int i, int x) {
        for (; i <= n - 1; i += i & -i) fen[i] ^= x;
    }

    static int fenQry(int i) {
        int r = 0;
        for (; i > 0; i -= i & -i) r ^= fen[i];
        return r;
    }

    static void pointXor(int pos, int x) {      // E[pos] ^= x
        if (pos > n - 1 || x == 0) return;
        E[pos] ^= x;
        fenUpd(pos, x);
        int p = pos + n;
        Arrays.fill(t, p * B, p * B + B, 0);
        ins(t, p * B, E[pos]);
        for (p >>= 1; p >= 1; p >>= 1) pull(p);
    }

    static void rangeXor(int l, int r, int x) {
        pointXor(l, x);
        pointXor(r + 1, x);
    }

    public static void main(String[] args) throws IOException {
        int N = (int) readLong();
        int Q = (int) readLong();
        n = N + 1;

        int[] A = new int[N + 2];
        for (int i = 1; i <= N; i++) A[i] = (int) readLong();

        E = new int[n];
        fen = new int[n + 1];
        t = new int[2 * n * B];

        for (int i = 1; i <= N; i++) {
            E[i] = A[i] ^ A[i - 1];
            ins(t, (i + n) * B, E[i]);
            fenUpd(i, E[i]);
        }
        for (int i = n - 1; i >= 1; i--) pull(i);

        StringBuilder sb = new StringBuilder();
        int[] R = new int[B];

        while (Q-- > 0) {
            char c = readChar();
            if (c == 'X') {
                int l = (int) readLong(), r = (int) readLong(), x = (int) readLong();
                rangeXor(l, r, x);
            } else if (c == 'S') {
                int i = (int) readLong(), x = (int) readLong();
                int d = fenQry(i) ^ x;
                if (d != 0) rangeXor(i, i, d);
            } else {
                int l = (int) readLong(), r = (int) readLong(), p = (int) readLong();
                long k = readLong();

                Arrays.fill(R, 0);
                if (l < r) {
                    int lo = l + 1 + n, hi = r + 1 + n;     // [lo, hi)
                    for (; lo < hi; lo >>= 1, hi >>= 1) {
                        if ((lo & 1) == 1) {
                            for (int i = 0; i < B; i++) {
                                int v = t[lo * B + i];
                                if (v != 0) ins(R, 0, v);
                            }
                            lo++;
                        }
                        if ((hi & 1) == 1) {
                            --hi;
                            for (int i = 0; i < B; i++) {
                                int v = t[hi * B + i];
                                if (v != 0) ins(R, 0, v);
                            }
                        }
                    }
                }

                // reduced row-echelon form
                for (int i = 0; i < B; i++) {
                    if (R[i] == 0) continue;
                    for (int j = i + 1; j < B; j++)
                        if (((R[j] >> i) & 1) == 1) R[j] ^= R[i];
                }

                int rank = 0;
                for (int i = 0; i < B; i++) if (R[i] != 0) rank++;

                if (k > (1L << rank)) {
                    sb.append("-1\n");
                    continue;
                }

                int a = 0;
                if (p == 1) {
                    a = fenQry(l);
                    for (int i = B - 1; i >= 0; i--)
                        if (R[i] != 0 && ((a >> i) & 1) == 1) a ^= R[i];
                }

                long m = k - 1;
                int ans = a, idx = 0;
                for (int i = 0; i < B; i++) {
                    if (R[i] != 0) {
                        if (((m >> idx) & 1L) == 1L) ans ^= R[i];
                        idx++;
                    }
                }
                sb.append(ans).append('\n');
            }
        }
        System.out.print(sb);
    }
}

import java.io.*;
import java.util.*;

public class Main {
    static final int MAXN = 200005;
    static int[] lc = new int[MAXN], rc = new int[MAXN], sz = new int[MAXN];
    static int[] val = new int[MAXN], sm = new int[MAXN], mn = new int[MAXN], mx = new int[MAXN];
    static boolean[] rv = new boolean[MAXN], iv = new boolean[MAXN];
    static int[] pri = new int[MAXN];
    static int sa, sb;
    static int qs, qmn;
    static Random rng = new Random(712367);

    static void applyInv(int t) {
        if (t == 0) return;
        val[t] = -val[t];
        sm[t] = -sm[t];
        int a = mn[t], b = mx[t];
        mn[t] = -b;
        mx[t] = -a;
        iv[t] = !iv[t];
    }

    static void applyRev(int t) {
        if (t == 0) return;
        int tmp = lc[t]; lc[t] = rc[t]; rc[t] = tmp;
        int a = mn[t], b = mx[t];
        mn[t] = sm[t] - b;
        mx[t] = sm[t] - a;
        rv[t] = !rv[t];
    }

    static void push(int t) {
        if (iv[t]) { applyInv(lc[t]); applyInv(rc[t]); iv[t] = false; }
        if (rv[t]) { applyRev(lc[t]); applyRev(rc[t]); rv[t] = false; }
    }

    static void pull(int t) {
        int l = lc[t], r = rc[t];
        sz[t] = sz[l] + sz[r] + 1;
        int s = sm[l] + val[t];
        sm[t] = s + sm[r];
        mn[t] = Math.min(mn[l], s + mn[r]);
        mx[t] = Math.max(mx[l], s + mx[r]);
    }

    static void split(int t, int k) {
        if (t == 0) { sa = sb = 0; return; }
        push(t);
        if (sz[lc[t]] >= k) {
            split(lc[t], k);
            lc[t] = sb;
            sb = t;
            pull(t);
        } else {
            split(rc[t], k - sz[lc[t]] - 1);
            rc[t] = sa;
            sa = t;
            pull(t);
        }
    }

    static int merge(int a, int b) {
        if (a == 0) return b;
        if (b == 0) return a;
        if (pri[a] > pri[b]) {
            push(a);
            rc[a] = merge(rc[a], b);
            pull(a);
            return a;
        } else {
            push(b);
            lc[b] = merge(a, lc[b]);
            pull(b);
            return b;
        }
    }

    // balanced build over [lo, hi], priorities decrease with depth (heap-valid)
    static int build(int lo, int hi, int depth) {
        if (lo > hi) return 0;
        int mid = (lo + hi) >>> 1;
        pri[mid] = (1 << 30) - depth * (1 << 20) + rng.nextInt(1 << 20);
        lc[mid] = build(lo, mid - 1, depth + 1);
        rc[mid] = build(mid + 1, hi, depth + 1);
        pull(mid);
        return mid;
    }

    // read-only range query on 0-based positions [l, r] inside subtree t
    static void query(int t, int l, int r) {
        if (t == 0 || l > r) return;
        if (l == 0 && r == sz[t] - 1) {
            qmn = Math.min(qmn, qs + mn[t]);
            qs += sm[t];
            return;
        }
        push(t);
        int ls = sz[lc[t]];
        if (l < ls) query(lc[t], l, Math.min(r, ls - 1));
        if (l <= ls && r >= ls) {
            qs += val[t];
            qmn = Math.min(qmn, qs);
        }
        if (r > ls) query(rc[t], Math.max(0, l - ls - 1), r - ls - 1);
    }

    static DataInputStream in = new DataInputStream(new BufferedInputStream(System.in, 1 << 16));
    static byte[] buf = new byte[1 << 16];
    static int bl = 0, bp = 0;

    static int rb() throws IOException {
        if (bp == bl) {
            bl = in.read(buf, 0, buf.length);
            bp = 0;
            if (bl <= 0) return -1;
        }
        return buf[bp++];
    }

    static int nonSpace() throws IOException {
        int c = rb();
        while (c == ' ' || c == '\n' || c == '\r' || c == '\t') c = rb();
        return c;
    }

    static long readLong() throws IOException {
        int c = nonSpace();
        long x = 0;
        while (c >= '0' && c <= '9') {
            x = x * 10 + (c - '0');
            c = rb();
        }
        return x;
    }

    public static void main(String[] args) throws IOException {
        int n = (int) readLong();
        int q = (int) readLong();
        for (int i = 1; i <= n; i++) {
            int c = nonSpace();
            val[i] = (c == '(') ? 1 : -1;
        }
        // pull needs children built first; build() calls pull after children,
        // but leaves need val set, which is done above.
        int root = build(1, n, 0);

        StringBuilder out = new StringBuilder();
        while (q-- > 0) {
            int c = nonSpace();
            int l = (int) readLong();
            int r = (int) readLong();
            int len = r - l + 1;
            if (c == 'Q') {
                if ((len & 1) == 1) {
                    out.append("-1\n");
                } else {
                    qs = 0;
                    qmn = 0;
                    query(root, l - 1, r - 1);
                    int A = -qmn;
                    int B = qs + A;
                    out.append((A + 1) / 2 + (B + 1) / 2).append('\n');
                }
                continue;
            }
            long k = 0;
            if (c == 'C') {
                k = readLong();
                if (k % len == 0) continue;
            }
            split(root, r);
            int ab = sa, b = sb;
            split(ab, l - 1);
            int a = sa, m = sb;
            if (c == 'R') {
                applyRev(m);
            } else if (c == 'F') {
                applyInv(m);
            } else {
                int kk = (int) (k % len);
                split(m, len - kk);
                int x = sa, y = sb;
                m = merge(y, x);
            }
            root = merge(merge(a, m), b);
        }
        System.out.print(out);
    }
}

#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
const ll MOD = 998244353, G = 3;

ll pw(ll b, ll e) {
    ll r = 1; b %= MOD;
    for (; e; e >>= 1, b = b * b % MOD) if (e & 1) r = r * b % MOD;
    return r;
}

void ntt(vector<ll>& a, bool inv) {
    int n = a.size();
    for (int i = 1, j = 0; i < n; i++) {
        int bit = n >> 1;
        for (; j & bit; bit >>= 1) j ^= bit;
        j ^= bit;
        if (i < j) swap(a[i], a[j]);
    }
    for (int len = 2; len <= n; len <<= 1) {
        ll w = pw(G, (MOD - 1) / len);
        if (inv) w = pw(w, MOD - 2);
        for (int i = 0; i < n; i += len) {
            ll cur = 1;
            for (int j = 0; j < len / 2; j++) {
                ll u = a[i + j], v = a[i + j + len / 2] * cur % MOD;
                a[i + j] = (u + v) % MOD;
                a[i + j + len / 2] = (u - v + MOD) % MOD;
                cur = cur * w % MOD;
            }
        }
    }
    if (inv) {
        ll ni = pw(n, MOD - 2);
        for (auto& x : a) x = x * ni % MOD;
    }
}

int main() {
    ll p, a; int Q;
    scanf("%lld %lld %d", &p, &a, &Q);

    // Legendre symbol table
    vector<int> chi(p, -1);
    chi[0] = 0;
    for (ll x = 1; x < p; x++) chi[x * x % p] = 1;

    // cnt[v] = #x with x^3 + a x == v (mod p)
    vector<int> cnt(p, 0);
    for (ll x = 0; x < p; x++) cnt[((x * x % p) * x + a * x) % p]++;

    int N = 1 << 19;
    vector<ll> A(N, 0), B(N, 0);
    for (ll i = 0; i < p; i++) A[i] = cnt[p - 1 - i];
    for (ll u = 0; u <= 2 * p - 2; u++) {
        int c = chi[u % p];
        B[u] = (c == -1) ? MOD - 1 : c;
    }

    ntt(A, false); ntt(B, false);
    for (int i = 0; i < N; i++) A[i] = A[i] * B[i] % MOD;
    ntt(A, true);

    string out;
    while (Q--) {
        ll b; scanf("%lld", &b);
        ll s = A[p - 1 + b];
        if (s > MOD / 2) s -= MOD;
        out += to_string(p + 1 + s);
        out += '\n';
    }
    fputs(out.c_str(), stdout);
}

#include <bits/stdc++.h>

// By Lucas' theorem, C(a, k) is odd  <=>  k is a submask of a  (k & a == k).

        for (int i = 0; i < n; i++) {
            unsigned long long a;
            scanf("%llu", &a);
            x ^= __builtin_popcountll(a);
        }
        puts(x ? "ALICE" : "BOB");
    }
    return 0;

const int MAXN = 200005;
const int LOG = 18;

int up[LOG][MAXN], dep[MAXN];
int par[MAXN], ea[MAXN], eb[MAXN], dm[MAXN];
vector<int> adj[MAXN];

struct Edge { int u, v; long long w; };
bool cmp(const Edge& a, const Edge& b) { return a.w > b.w; }

int findp(int x) {
    while (par[x] != x) { par[x] = par[par[x]]; x = par[x]; }
    return x;
}

int lca(int u, int v) {
    if (dep[u] < dep[v]) swap(u, v);
    int diff = dep[u] - dep[v];
    for (int k = 0; k < LOG; k++)
        if ((diff >> k) & 1) u = up[k][u];
    if (u == v) return u;
    for (int k = LOG - 1; k >= 0; k--)
        if (up[k][u] != up[k][v]) { u = up[k][u]; v = up[k][v]; }
    return up[0][u];
}

int dist(int u, int v) {
    return dep[u] + dep[v] - 2 * dep[lca(u, v)];
}

            up[k][i] = up[k - 1][up[k - 1][i]];

struct Line {
    ll m, b;
    ll eval(ll x) const { return m * x + b; }
};

int M;
vector<ll> xs;
vector<Line> tr;
vector<char> has;

void insertLine(int node, int lo, int hi, Line ln) {
    if (!has[node]) {
        tr[node] = ln;
        has[node] = 1;
        return;
    }
    int mid = (lo + hi) / 2;
    if (ln.eval(xs[mid]) < tr[node].eval(xs[mid])) swap(ln, tr[node]);
    if (lo == hi) return;
    if (ln.eval(xs[lo]) < tr[node].eval(xs[lo]))
        insertLine(node * 2, lo, mid, ln);
    else if (ln.eval(xs[hi]) < tr[node].eval(xs[hi]))
        insertLine(node * 2 + 1, mid + 1, hi, ln);
}

ll query(int pos) {
    ll x = xs[pos];
    ll best = LLONG_MAX;
    int node = 1, lo = 0, hi = M - 1;
    while (true) {
        if (has[node]) best = min(best, tr[node].eval(x));
        if (lo == hi) break;
        int mid = (lo + hi) / 2;
        if (pos <= mid) { node = node * 2; hi = mid; }
        else            { node = node * 2 + 1; lo = mid + 1; }
    }
    return best;
}

    int n;
    ll C;
    scanf("%d %lld", &n, &C);
    vector<ll> P(n + 1, 0);

        scanf("%lld", &a);
        P[i] = P[i - 1] + a;
    }

    xs.erase(unique(xs.begin(), xs.end()), xs.end());
    M = xs.size();
    tr.assign(4 * M + 4, Line{0, 0});
    has.assign(4 * M + 4, 0);

    // j = 0: dp[0] = 0, P[0] = 0
    insertLine(1, 0, M - 1, Line{0, 0});

    ll dp = 0;
    for (int i = 1; i <= n; i++) {
        int pos = lower_bound(xs.begin(), xs.end(), P[i]) - xs.begin();

typedef long long ll;

int main() {
    int M, N, K;
    if (scanf("%d %d %d", &M, &N, &K) != 3) return 0;
    vector<ll> g((size_t)M * N);
    for (auto &x : g) scanf("%lld", &x);

    if (g[0] < 0) { puts("-1"); return 0; }

    const ll NEG = LLONG_MIN / 4;          // marks an unreachable state
    const int last = M * N - 1;

    vector<ll> dp(M * N), init(M * N, NEG), nxt(M * N);
    vector<ll> colMax(N), rowMax(M);
    init[0] = g[0];                        // layer 0: start cell only
    ll ans = NEG;

    for (int k = 0; k <= K; k++) {
        // Fill layer k using right/down moves plus arrivals by jump (init)
        for (int r = 0; r < M; r++) {
            for (int c = 0; c < N; c++) {
                int id = r * N + c;
                ll cand = NEG;
                if (r > 0) cand = max(cand, dp[id - N]);
                if (c > 0) cand = max(cand, dp[id - 1]);
                ll v = init[id];
                if (cand >= 0) {
                    ll t = cand + g[id];
                    if (t >= 0) v = max(v, t);
                }
                dp[id] = v;
            }
        }
        ans = max(ans, dp[last]);
        if (k == K) break;

        // Best energy per column / row (exit excluded: the route ends there)
        fill(colMax.begin(), colMax.end(), NEG);
        fill(rowMax.begin(), rowMax.end(), NEG);
        for (int r = 0; r < M; r++) {
            for (int c = 0; c < N; c++) {
                int id = r * N + c;
                if (id == last) continue;
                colMax[c] = max(colMax[c], dp[id]);
                rowMax[r] = max(rowMax[r], dp[id]);
            }
        }

        // Build arrivals for layer k+1
        for (int r = 0; r < M; r++) {
            for (int c = 0; c < N; c++) {
                int id = r * N + c;
                ll best = NEG;
                if (c + 1 < N && colMax[c + 1] >= 0)   // jump from column c+1 to column c
                    best = max(best, colMax[c + 1] + g[id]);
                if (r + 1 < M && rowMax[r + 1] >= 0)   // jump from row r+1 to row r
                    best = max(best, rowMax[r + 1] + g[id]);
                nxt[id] = (best >= 0) ? best : NEG;
            }
        }
        init.swap(nxt);
    }

    if (ans < 0) puts("-1");
    else printf("%lld\n", ans);
    return 0;
}

typedef long long ll;
typedef unsigned long long ull;
const ll MOD = 1000000007LL;

typedef vector<vector<ll>> Mat;

Mat mul(const Mat& A, const Mat& B) {
    int n = A.size();
    Mat C(n, vector<ll>(n, 0));
    for (int i = 0; i < n; i++)
        for (int k = 0; k < n; k++) {
            if (A[i][k] == 0) continue;
            ll a = A[i][k];
            for (int j = 0; j < n; j++)
                C[i][j] = (C[i][j] + a * B[k][j]) % MOD;
        }
    return C;
}

int idx(char c) {
    switch (c) {
        case 'A': return 0;
        case 'C': return 1;
        case 'G': return 2;
        default:  return 3;
    }
}

int main() {
    int n;
    ll L;
    cin >> n >> L;

    // Trie
    vector<array<int, 4>> go(1, {-1, -1, -1, -1});
    vector<int> fail(1, 0);
    vector<bool> bad(1, false);

    for (int i = 0; i < n; i++) {
        string s;
        cin >> s;
        int cur = 0;
        for (char ch : s) {
            int c = idx(ch);
            if (go[cur][c] == -1) {
                go[cur][c] = go.size();
                go.push_back({-1, -1, -1, -1});
                fail.push_back(0);
                bad.push_back(false);
            }
            cur = go[cur][c];
        }
        bad[cur] = true;
    }

    // Build failure links + full automaton (BFS)
    queue<int> q;
    for (int c = 0; c < 4; c++) {
        if (go[0][c] == -1) go[0][c] = 0;
        else {
            fail[go[0][c]] = 0;
            q.push(go[0][c]);
        }
    }
    while (!q.empty()) {
        int u = q.front(); q.pop();
        bad[u] = bad[u] || bad[fail[u]];
        for (int c = 0; c < 4; c++) {
            int v = go[u][c];
            if (v == -1) {
                go[u][c] = go[fail[u]][c];
            } else {
                fail[v] = go[fail[u]][c];
                q.push(v);
            }
        }
    }

    int S = go.size();

    // Transition matrix (only safe states)
    Mat M(S, vector<ll>(S, 0));
    for (int i = 0; i < S; i++) {
        if (bad[i]) continue;
        for (int c = 0; c < 4; c++) {
            int j = go[i][c];
            if (!bad[j]) M[i][j] += 1;
        }
    }

    // Result = start vector * M^L ; keep as vector, square the matrix
    vector<ll> vec(S, 0);
    vec[0] = 1;

    while (L > 0) {
        if (L & 1) {
            vector<ll> nv(S, 0);
            for (int i = 0; i < S; i++) {
                if (vec[i] == 0) continue;
                for (int j = 0; j < S; j++)
                    nv[j] = (nv[j] + vec[i] * M[i][j]) % MOD;
            }
            vec = nv;
        }
        L >>= 1;
        if (L) M = mul(M, M);
    }

    ll ans = 0;
    for (int i = 0; i < S; i++)
        if (!bad[i]) ans = (ans + vec[i]) % MOD;

    cout << ans << endl;
    return 0;

    int d = dep[a] - dep[b];
    for (int k = 0; k <= LOG; k++)
        if (d >> k & 1) a = up[a][k];
    if (a == b) return a;
    for (int k = LOG; k >= 0; k--)

    int timer = 0;

    stack_.push_back(1);

            st[w] = timer; ord[timer++] = w;
            stack_.push_back(w);
        } else {
            en[u] = timer; ord[timer++] = u;
            stack_.pop_back();
        }

            if (++cnt[c] == 2) cur++;

vector<ll> bit;
void upd(int i, ll v) { for (; i <= M; i += i & -i) bit[i] += v; }
ll qry(int i) { ll s = 0; for (; i > 0; i -= i & -i) s += bit[i]; return s; }

int main() {
    int n, q;
    scanf("%d %d", &n, &q);
    vector<array<ll,3>> pts(n);              // x, y, w
    vector<ll> ys(n);

        scanf("%lld %lld %lld", &pts[i][0], &pts[i][1], &pts[i][2]);
        ys[i] = pts[i][1];
    }
    sort(pts.begin(), pts.end());
    sort(ys.begin(), ys.end());
    ys.erase(unique(ys.begin(), ys.end()), ys.end());
    M = ys.size();
    bit.assign(M + 1, 0);

    struct Ev { ll x; int id, sign, lo, hi; };
    vector<Ev> ev;
    ev.reserve(2 * q);
    vector<ll> ans(q, 0);

    for (int j = 0; j < q; j++) {
        ll x1, x2, y1, y2;
        scanf("%lld %lld %lld %lld", &x1, &x2, &y1, &y2);
        if (x1 > x2 || y1 > y2) continue;    // empty zone -> answer 0
        // closed zone: x1 <= x <= x2, y1 <= y <= y2
        int lo = lower_bound(ys.begin(), ys.end(), y1) - ys.begin();  // count of ys < y1
        int hi = upper_bound(ys.begin(), ys.end(), y2) - ys.begin();  // count of ys <= y2
        if (lo >= hi) continue;
        ev.push_back({x2,     j, +1, lo, hi});
        ev.push_back({x1 - 1, j, -1, lo, hi});
    }
    sort(ev.begin(), ev.end(), [](const Ev& a, const Ev& b){ return a.x < b.x; });

    int p = 0;
    for (auto& e : ev) {
        while (p < n && pts[p][0] <= e.x) {
            int yi = lower_bound(ys.begin(), ys.end(), pts[p][1]) - ys.begin() + 1;
            upd(yi, pts[p][2]);
            p++;
        }

#include <bits/stdc++.h>
using namespace std;
typedef long long ll;

int n, K;
vector<ll> pref;
vector<ll> prv, cur;

inline ll cost(int j, int i) {
    ll s = pref[i] - pref[j];
    ll l = i - j;
    return s * s + l * l;
}

// compute cur[i] for i in [lo, hi], optimal j in [optlo, opthi]
void solve(int lo, int hi, int optlo, int opthi) {
    if (lo > hi) return;
    int mid = (lo + hi) / 2;
    ll best = LLONG_MAX;
    int bestj = optlo;
    int up = min(mid - 1, opthi);
    for (int j = optlo; j <= up; j++) {
        if (prv[j] == LLONG_MAX) continue;
        ll v = prv[j] + cost(j, mid);
        if (v < best) { best = v; bestj = j; }
    }
    cur[mid] = best;
    solve(lo, mid - 1, optlo, bestj);
    solve(mid + 1, hi, bestj, opthi);
}

int main() {
    scanf("%d %d", &n, &K);
    pref.assign(n + 1, 0);
    for (int i = 1; i <= n; i++) {
        ll a; scanf("%lld", &a);
        pref[i] = pref[i - 1] + a;
    }
    prv.assign(n + 1, LLONG_MAX);
    cur.assign(n + 1, LLONG_MAX);
    prv[0] = 0;
    for (int k = 1; k <= K; k++) {
        fill(cur.begin(), cur.end(), LLONG_MAX);
        solve(k, n, k - 1, n - 1);
        swap(prv, cur);
    }
    printf("%lld\n", prv[n]);
    return 0;
}

#include <iostream>
#include <vector>
#include <deque>
#include <algorithm>

using namespace std;

int main() {
    // Optimize standard I/O operations for performance
    ios_base::sync_with_stdio(false);
    cin.tie(NULL);

    int n;
    long long k;
    
    if (!(cin >> n >> k)) return 0;

    vector<long long> a(n);
    for (int i = 0; i < n; ++i) {
        cin >> a[i];
    }

    // P[i] will store the sum of elements from A[0] to A[i-1]
    vector<long long> p(n + 1, 0);
    for (int i = 0; i < n; ++i) {
        p[i + 1] = p[i] + a[i];
    }

    deque<int> dq;
    int min_length = n + 1; // Initialize with a value strictly greater than max possible length

    for (int i = 0; i <= n; ++i) {
        // 1. Check if we can form a valid subarray >= K using the front of the deque
        while (!dq.empty() && p[i] - p[dq.front()] >= k) {
            min_length = min(min_length, i - dq.front());
            dq.pop_front(); // We found the shortest valid subarray starting at dq.front()
        }

        // 2. Maintain the monotonic property (increasing prefix sums)
        while (!dq.empty() && p[i] <= p[dq.back()]) {
            dq.pop_back(); // Remove larger prefix sums as they are worse starting points
        }

        // 3. Add the current index as a potential future starting point
        dq.push_back(i);
    }

    // Output the result
    if (min_length == n + 1) {
        cout << -1 << "\n";
    } else {
        cout << min_length << "\n";
    }

    return 0;
}
